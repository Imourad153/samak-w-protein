let cart = [];

function toggleCart() {
    document.getElementById("cart").classList.toggle("active");
}

function addToCart(name, price) {

    let item = cart.find(product => product.name === name);

    if (item) {
        item.qty++;
    } else {
        cart.push({
            name: name,
            price: price,
            qty: 1
        });
    }

    updateCart();
}

function removeItem(index) {

    cart.splice(index,1);

    updateCart();

}

function updateCart(){

    const cartItems=document.getElementById("cart-items");

    const totalPrice=document.getElementById("total-price");

    const cartCount=document.getElementById("cart-count");

    cartItems.innerHTML="";

    let total=0;

    let count=0;

    if(cart.length===0){

        cartItems.innerHTML="<p class='empty'>لا توجد منتجات داخل السلة</p>";

    }

    cart.forEach((item,index)=>{

        total+=item.price*item.qty;

        count+=item.qty;

        cartItems.innerHTML+=`

        <div class="cart-item">

            <div>

                <strong>${item.name}</strong>

                <br>

                ${item.qty} × ${item.price} جنيه

            </div>

            <button onclick="removeItem(${index})">

                حذف

            </button>

        </div>

        `;

    });

    totalPrice.innerHTML=total;

    cartCount.innerHTML=count;

}

document.getElementById("checkoutBtn").onclick=function(){

    if(cart.length===0){

        alert("السلة فارغة");

        return;

    }

    let message="السلام عليكم%0A";

    message+="أرغب فى طلب:%0A%0A";

    let total=0;

    cart.forEach(item=>{

        message+=`${item.name} × ${item.qty}%0A`;

        total+=item.price*item.qty;

    });

    message+=`%0Aالإجمالى : ${total} جنيه`;

    window.open("https://wa.me/201159866529?text="+message,"_blank");

};

updateCart();
